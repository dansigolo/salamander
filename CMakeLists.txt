cmake_minimum_required(VERSION 3.21)
project(OpenSalamanderQt VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    add_compile_options(/utf-8)
endif()
include(CTest)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Concurrent)
add_library(salamander_core port/core/mask.cpp)
target_include_directories(salamander_core PUBLIC port/core)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The native Qt port currently targets Linux.")
endif()
add_library(salamander_operations port/operations/fileoperations_linux.cpp port/operations/operationcontroller.cpp port/operations/copyoperations_linux.cpp port/operations/copycontroller.cpp)
set_target_properties(salamander_operations PROPERTIES AUTOMOC ON)
target_include_directories(salamander_operations PUBLIC port/operations)
target_link_libraries(salamander_operations PUBLIC Qt6::Core Qt6::Concurrent)
add_library(salamander_ui port/qt/filepanel.cpp port/qt/fileview.cpp port/qt/detailedfilemodel.cpp port/qt/mainwindow.cpp port/qt/namedialog.cpp port/qt/copydialogs.cpp)
set_target_properties(salamander_ui PROPERTIES AUTOMOC ON)
target_include_directories(salamander_ui PUBLIC port/qt)
target_link_libraries(salamander_ui PUBLIC Qt6::Widgets Qt6::Concurrent salamander_operations)
add_executable(salamander-qt port/qt/main.cpp)
target_link_libraries(salamander-qt PRIVATE salamander_ui)
if(BUILD_TESTING)
    find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
    add_executable(core_tests port/tests/core_tests.cpp)
    target_link_libraries(core_tests PRIVATE salamander_core)
    add_test(NAME core_masks COMMAND core_tests)
    add_executable(operation_tests port/tests/operation_tests.cpp)
    set_target_properties(operation_tests PROPERTIES AUTOMOC ON)
    target_link_libraries(operation_tests PRIVATE salamander_operations Qt6::Test)
    add_test(NAME file_operations COMMAND operation_tests)
    set_tests_properties(file_operations PROPERTIES TIMEOUT 30)
    add_executable(copy_tests port/tests/copy_tests.cpp)
    set_target_properties(copy_tests PROPERTIES AUTOMOC ON)
    target_link_libraries(copy_tests PRIVATE salamander_operations Qt6::Test)
    add_test(NAME file_copy COMMAND copy_tests)
    set_tests_properties(file_copy PROPERTIES TIMEOUT 45)
    add_executable(move_tests port/tests/move_tests.cpp)
    set_target_properties(move_tests PROPERTIES AUTOMOC ON)
    target_link_libraries(move_tests PRIVATE salamander_operations Qt6::Test)
    add_test(NAME file_move COMMAND move_tests)
    set_tests_properties(file_move PROPERTIES TIMEOUT 60)
    add_executable(panel_tests port/tests/panel_tests.cpp)
    set_target_properties(panel_tests PROPERTIES AUTOMOC ON)
    target_link_libraries(panel_tests PRIVATE salamander_ui Qt6::Test)
    add_test(NAME qt_panels COMMAND panel_tests)
    add_executable(window_tests port/tests/window_tests.cpp)
    set_target_properties(window_tests PROPERTIES AUTOMOC ON)
    target_link_libraries(window_tests PRIVATE salamander_ui Qt6::Test)
    add_test(NAME qt_window COMMAND window_tests)
    set_tests_properties(qt_panels qt_window PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
endif()
