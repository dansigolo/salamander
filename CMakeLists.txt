cmake_minimum_required(VERSION 3.21)
project(OpenSalamanderQt VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
add_library(salamander_core port/core/mask.cpp)
target_include_directories(salamander_core PUBLIC port/core)
add_library(salamander_ui port/qt/filepanel.cpp)
target_include_directories(salamander_ui PUBLIC port/qt)
target_link_libraries(salamander_ui PUBLIC Qt6::Widgets)
add_executable(salamander-qt port/qt/main.cpp)
target_link_libraries(salamander-qt PRIVATE salamander_ui)
if(BUILD_TESTING)
    find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
    add_executable(core_tests port/tests/core_tests.cpp)
    target_link_libraries(core_tests PRIVATE salamander_core)
    add_test(NAME core_masks COMMAND core_tests)
    add_executable(panel_tests port/tests/panel_tests.cpp)
    set_target_properties(panel_tests PROPERTIES AUTOMOC ON)
    target_link_libraries(panel_tests PRIVATE salamander_ui Qt6::Test)
    add_test(NAME qt_panels COMMAND panel_tests)
    set_tests_properties(qt_panels PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
endif()
